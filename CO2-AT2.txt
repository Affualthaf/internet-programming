    <script>


        /* ==========================================
           MAIN RESPONSIVE NAVIGATION
        ========================================== */

        const menuButton =
            document.getElementById("menuButton");

        const navLinks =
            document.getElementById("navLinks");


        menuButton.addEventListener(
            "click",
            function () {

                navLinks.classList.toggle("active");


                /*
                    Update accessibility attribute
                */

                const isOpen =
                    navLinks.classList.contains("active");

                menuButton.setAttribute(
                    "aria-expanded",
                    isOpen
                );

            }
        );


        /* ==========================================
           CLOSE MENU AFTER CLICKING A LINK
        ========================================== */

        const links =
            document.querySelectorAll(
                "#navLinks a"
            );


        links.forEach(function(link) {

            link.addEventListener(
                "click",
                function() {

                    navLinks.classList.remove("active");

                    menuButton.setAttribute(
                        "aria-expanded",
                        "false"
                    );

                }
            );

        });



        /* ==========================================
           DEMO NAVIGATION MENU

        function toggleDemoMenu() {

            const demoLinks =

           DYNAMIC MENU
        ========================================== */

        const dynamicMenuButton =
            document.getElementById(
                "dynamicMenuButton"
            );

        const dynamicMenuList =
            document.getElementById(
                "dynamicMenuList"
            );


        dynamicMenuButton.addEventListener(
            "click",
            function() {


                /*
                    Toggle menu visibility
                */

                dynamicMenuList.classList.toggle(
                    "active"
                );


                /*
                    Check whether menu is open
                */

                const isOpen =
                    dynamicMenuList.classList.contains(
                        "active"
                    );


                /*
                    Change button text
                */

                if (isOpen) {

                    dynamicMenuButton.textContent =
                        "✕ Close Menu";

                } else {

                    dynamicMenuButton.textContent =
                        "☰ Open Menu";

                }


                /*
                    Update ARIA attribute
                */

                dynamicMenuButton.setAttribute(
                    "aria-expanded",
                    isOpen
                );

            }
        );

    </script>
